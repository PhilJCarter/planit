90def _select_user_eos(slot, eosname=None, eosdir=None): 

91 """Load or return a custom EOS slot without hiding invalid requests.""" 

92 cache_name = _USER_EOS_CACHE_NAMES[slot] 

93 cached_eos = globals()[cache_name] 

94 has_eosname = eosname is not None 

95 has_eosdir = eosdir is not None 

96 

97 if has_eosname != has_eosdir: 

98 raise ValueError( 

99 f'Custom EOS {slot} requires both eosname and eosdir; ' 

100 'provide neither only to retrieve an already loaded table.' 

101 ) 

102 

103 if not has_eosname: 

104 if cached_eos is None: 

105 raise ValueError( 

106 f'Custom EOS {slot} is not loaded. Provide both eosname and eosdir.' 

107 ) 

108 return cached_eos 

109 

110 # Assign only after the loader succeeds, so a malformed replacement cannot 

111 # leave a slot uninitialised or make select() silently return stale data. 

112 new_eos = loadANEOSEOS( 

113 eos=eosname, 

114 eostype='SESAME', 

115 eosdir=eosdir, 

116 user=True, 

117 womaID=USER_EOS_SLOTS[slot], 

118 ) 

119 globals()[cache_name] = new_eos 

120 return new_eos

123def select(name, eosname=None, eosdir=None): 

124 """Return an EOS table, loading it if necessary. 

125 

126 Bundled EOS tables can be selected by their established names or material 

127 IDs. Custom SESAME tables use one of the five ``User0``--``User4`` slots 

128 (or their IDs 900--904) and must be loaded with both ``eosname`` and 

129 ``eosdir``:: 

130 

131 table = select('User0', eosname='MyMaterial', eosdir='/path/to/table') 

132 

133 ``eosdir`` may be a string or path-like object and does not need a trailing 

134 slash. Repeating this call with both arguments replaces that slot after a 

135 successful load. Calling ``select('User0')`` later returns the cached 

136 table; calling it before a successful load, or supplying only one of the 

137 two arguments, raises ``ValueError``. 

138 """ 

139 user_slot = _user_slot_name(name) 

140 if user_slot is not None: 

141 return _select_user_eos(user_slot, eosname=eosname, eosdir=eosdir) 

142 if isinstance(name, str) and name.startswith('User'): 

143 raise ValueError( 

144 f'Unknown user EOS slot {name!r}. Supported slots are User0 through User4 ' 

145 '(WoMa IDs 900 through 904).' 

146 ) 

147 

148 if name in ironnames: 

149 global ANEOSIron 

150 if not ANEOSIron: 

151 ANEOSIron = loadEOS(eos='Iron-ANEOS-SLVTv0.2G1', eostype='ANEOS') 

152 return ANEOSIron 

153 elif name in alloynames: 

154 global ANEOSFeSiAlloy 

155 if not ANEOSFeSiAlloy: 

156 ANEOSFeSiAlloy = loadEOS(eos='Fe85Si15-ANEOS-SLVTv0.2G1', eostype='ANEOS') 

157 return ANEOSFeSiAlloy 

158 elif name in forsteritenames: 

159 global ANEOSForsterite 

160 if not ANEOSForsterite: 

161 ANEOSForsterite = loadEOS(eos='Forsterite-ANEOS-SLVTv1.0G1', eostype='ANEOS') 

162 return ANEOSForsterite 

163 elif name in pyrolitenames: 

164 global ANEOSPyrolite 

165 if not ANEOSPyrolite: 

166 ANEOSPyrolite = loadEOS(eos='Pyrolite_ANEOS_SLVTv0.2', eostype='ANEOS') 

167 return ANEOSPyrolite 

168 elif name in aquawaternames: 

169 global AQUAWater 

170 if not AQUAWater: 

171 AQUAWater = loadEOS(eos='Water-AQUA-v1.0', eostype='AQUA') 

172 return AQUAWater 

173 elif name in fivephasewaternames: 

174 global FivePhaseWater 

175 if not FivePhaseWater: 

176 FivePhaseWater = loadEOS(eos='5PhaseEOSv8.3', eostype='SESAME') 

177 return FivePhaseWater 

178 elif name in hm80HHenames: 

179 global HM80HHe 

180 if not HM80HHe: 

181 HM80HHe = loadEOS(eos='HM80-HHe-v2.0', eostype='HM80') 

182 return HM80HHe 

183 else: 

184 raise ValueError('Unknown EOS:', name)

187class isentrope_class(eos_isentrope_class): 

188 """Class to hold isentrope data extracted from EOS table. 

189  

190 extract(material,entropy) - extract isentrope from EOS 

191  

192 """ 

193 def __init__(self, entropy=None, material=None): 

194 """A function to initialize the class object.""" 

195 eos_isentrope_class.__init__(self) 

196 self.entropy = entropy 

197 self.material = material 

198 self.intenergy = [] 

199 if self.material and self.entropy: 

200 self.extract()

202 def extract(self,material=None,entropy=None): 

203 """Extract isentrope at entropy from EOS specified by material""" 

204 if not self.material: 

205 if material: 

206 self.material = material 

207 else: 

208 print('error: no material specified') 

209 return 

210 if not self.entropy: 

211 if entropy: 

212 self.entropy = entropy 

213 else: 

214 print('error: entropy not specified') 

215 return 

216 EOS = select(self.material) 

217 self.density = EOS.rho 

218 self.ND = EOS.ND 

219 

220 # loop across all densities and extract the values for the requested isentrope 

221 for i in range(0,self.ND): 

222 ind = npy.where(EOS.S[:,i] > 0)[0] 

223 interpfunction = interpolate.interp1d(EOS.S[ind,i],EOS.P[ind,i]) # MJ/K/kg, GPa 

224 self.pressure = npy.append(self.pressure,interpfunction(self.entropy/1.E3)) # GPa 

225 interpfunction = interpolate.interp1d(EOS.S[ind,i],EOS.T[ind]) # MJ/K/kg, GPa 

226 self.temperature = npy.append(self.temperature,interpfunction(self.entropy/1.E3)) # GPa

256def calcprop(Qlab,Xlab,Ylab,X,Y,mats): 

257 """Calculate thermodynamic property 

258  

259 Qlab - label of property to calculate 

260 Xlab - label of 1st known property to calculate from 

261 Ylab - label of 2nd known property to calculate from 

262 X - array of 1st known property values 

263 X - array of 2nd known property values 

264 mats - array of material identifiers 

265  

266 returns array of interpolated property at X, Y points 

267 """ 

268 if npy.ndim(X) == 0: 

269 X = npy.array([X,]) 

270 Y = npy.array([Y,]) 

271 Z = npy.array([mats,]) 

272 

273 if not len(X)==len(Y)==len(mats): 

274 raise ValueError('X, Y, and mats arrays must be the same size/shape') 

275 

276 if Ylab == 'rho' and Xlab in ['T','U','S']: 

277 tmp = Y 

278 tmplab = Ylab 

279 Y = X 

280 Ylab = Xlab 

281 X = tmp 

282 Xlab = tmplab 

283 if not ( Xlab == 'rho' and Ylab in ['T','U','S']): 

284 raise NotImplementedError('Calculation of', Qlab, 'from', Xlab, 'and', Ylab, 'is not available' ) 

285 

286 if Qlab == 'P': 

287 uconversion_Q = uconversion_P_inv 

288 elif Qlab == 'T': 

289 uconversion_Q = 1. 

290 elif Qlab == 'U': 

291 uconversion_Q = uconversion_U_inv 

292 elif Qlab == 'S': 

293 uconversion_Q = uconversion_S_inv 

294 elif Qlab == 'cs': 

295 uconversion_Q = 1. 

296 else: 

297 raise NotImplementedError('Error: calculation of', Qlab, 'not supported.') 

298 return None 

299 

300 if Ylab == 'S': 

301 Y = Y*uconversion_S 

302 #elif Ylab == 'rho': 

303 # Y = Y 

304 elif Ylab == 'P': 

305 Y = Y*uconversion_P 

306 elif Ylab == 'U': 

307 Y = Y*uconversion_U 

308 

309 EOSlist = npy.empty(len(X),dtype=object) 

310 for mat in npy.unique(mats): 

311 EOS = select(mat) 

312 passer = EOS.make_passer_class() 

313 EOSlist = npy.where(mats==mat,passer,EOSlist) 

314 

315 Q = _calc_prop(Qlab,Xlab,Ylab,X,Y,EOSlist.tolist()) 

316 return Q*uconversion_Q

321 #print(Qlab,Xlab,Ylab) 

322 Q = npy.zeros(len(X)) 

323 for i in numba.prange(len(X)):

330 elif Ylab == 'T':

336 Q[i] = npy.nan 

337 else:
